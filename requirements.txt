# Lista de bibliotecas que o Render instala antes de ligar o servidor
fastapi
# uvicorn é quem "liga" o FastAPI e escuta os pedidos pela internet
uvicorn[standard]
# SQLAlchemy: deixa a gente falar com o banco usando classes Python em vez de SQL puro
sqlalchemy>=2.0
# psycopg: o "driver" que conversa com o Postgres (versão binária, não precisa compilar)
psycopg[binary]
# argon2-cffi: faz o HASH das senhas (padrão atual recomendado)
argon2-cffi
# PyJWT: cria e confere os tokens de login
pyjwt
# email-validator: o FastAPI usa para conferir se um e-mail tem formato válido
email-validator
# httpx: faz chamadas HTTPS a outros serviços (usamos para falar com a API do Brevo)
httpx
